        public Image GetImage(string strEncoded)
        {
            byte[] arrBytImage = Convert.FromBase64String(strEncoded);
            Image objImage = null;
            // Convierte un array de bytes a una imagen
            using (System.IO.MemoryStream stmBitmap = new System.IO.MemoryStream(arrBytImage, 0, arrBytImage.Length))
            {
                // Escribe el array de bytes en el stream de memoria
                stmBitmap.Write(arrBytImage, 0, arrBytImage.Length);
                // Carga la imagen del stream
                objImage = Image.FromStream(stmBitmap);
                // Cierra el stream
                stmBitmap.Close();
            }
            // Devuelve la imagen
            return objImage;
        }



        List<Delito_DTO> ConvertirAListaDeNumeros(string valor)
        {
            // Dividir el string por la coma
            string[] partes = valor.Split(',');
            List<Delito_DTO> numeros = new List<Delito_DTO>();

            // Crear un objeto Numero por cada parte y añadirlo a la lista
            foreach (string parte in partes)
            {
                numeros.Add(new Delito_DTO { ID_DELITO = int.Parse(parte) });
            }

            return numeros;
        }


        [HttpPost]
        public async Task<ActionResult> CreateAjax(Criminalidad_DTO Datos)
        {
            try
            {

                string ListaResponsables = null;
                if (Datos.DELITO_PRINCIPAL == 0)
                {
                    return Json(new { ok = false, mensaje = "Seleccione el delito principal !!!" });
                }

                if (Datos.ARRAY_DELITO[0] != "")
                {
                    for (int i = 0; i < Datos.ARRAY_DELITO.Length; ++i)
                    {
                        ListaResponsables += Datos.ARRAY_DELITO[i];
                        Datos.delito_DTO = ConvertirAListaDeNumeros(ListaResponsables);
                    }
                }
               
                if(Datos.FOTO_EXPENDIO != null)
                {
                    Datos.ID_CRIMINALIDAD_FOTOS = Guid.NewGuid().ToString();
                    Datos.tipo_doc = Path.GetExtension(Datos.FOTO_EXPENDIO.FileName);
                    Datos.ruta = SubirAdjunto(Datos.FOTO_EXPENDIO, Datos.tipo_doc, Datos.ID_CRIMINALIDAD_FOTOS);
                    Datos.servidor = System.Configuration.ConfigurationManager.AppSettings["AdjuntosPMU"] + "IRISP1/EXPENDIOSWEB";
                    Datos.name_file = Datos.FOTO_EXPENDIO.FileName;
                }
                

                if (Convert.ToInt32(Datos.ID_UNIDAD) == 0 || Datos.ID_ZONA == null || Datos.IDENTIFICACION_INFORMA == null ||
                    Datos.CELULAR == null || Datos.ID_TIPO_SERVICIO == null || Datos.ID_ESPECIALIDAD == null || Datos.ID_CLASE == null ||
                    Datos.NOMBRE_CLASE == null || Datos.ENTORNO_AFECTADO == null ||
                    Datos.ID_FUENTE == null || Datos.CARACTERISTICAS_GENERALES == null || Datos.NOMBRE_ENTORNO_AFECTADO == null) 

                    return Json(new { ok = false, mensaje = "Faltan campos de información que son obligatorios y no pueden estar vacíos !!!" });

                if (!await new PersonaRepositorio().ConsultarFuncionarioPorIdentificacion(Convert.ToInt64(Datos.IDENTIFICACION_INFORMA)))
                    return Json(new { ok = false, mensaje = "La identificacion suministrada no corresponde a un funcionario, la cual es obligatoria para la creación del IRISP1" });



                if (Datos.ARRAY_DELITO.Count() == 0)
                {
                    return Json(new { ok = false, mensaje = "Seleccione el delito principal !!!" });
                }
                

                if (string.IsNullOrEmpty(Datos.LONGITUD) && string.IsNullOrEmpty(Datos.LATITUD))
                    return Json(new { ok = false, mensaje = "La latitud y longitud no pueden estar vacias, seleccione un punto geográfico valido con coordenadas" });

                if (Datos.CODIGO_DANE == null && Datos.MUNICIPIO == null)
                    return Json(new { ok = false, mensaje = "La ubicación debe ser dentro del territorio Colombiano !!!" });

                Datos.IDENTIFICACION_CREACION = Convert.ToInt32(gestionClaims.Identificacion);
                Datos.MAQUINA_CREACION = Request.UserHostAddress;

                if (ModelState.IsValid)
                {
                    if (await new CriminalidadRepositorio().Agregar(Datos))
                    {
                        if (true)
                        {
                            //var ExitoCorreo = new CorreoRepositorio().EnvioCorreoBasico(Convert.ToInt64(Datos.IDENTIFICACION_INFORMA), "Notificación IRISP1.", Convert.ToInt64(Datos.IDENTIFICACION_INFORMA), "Se creó un caso IRISP1 con código de radicado: " + Datos.CODIGO + ". Para ver detalles de su caso, ingrese al sistema de información IRISP1.");
                            return Json(new { ok = true, mensaje = "Se agregó correctamente el registro " + Datos.CODIGO });
                        }
                    }
                }
                    

                string listaErrores = null;
                int contadorErrores = 0;

                foreach (var key in ModelState)
                {
                    if (key.Value.Errors.Count > 0)
                    {
                        contadorErrores++;
                        listaErrores += $" {contadorErrores}. {key.Value.Errors[0].ErrorMessage}";
                    }
                }

                if (string.IsNullOrEmpty(listaErrores))
                    return Json(new { ok = false, mensaje = "No se pudo guardar el registro" });
                else
                    return Json(new { ok = false, mensaje = listaErrores });
            }
            catch (Exception ex)
            {
                return await CapturaError("POST", "Irisp", "Criminalidad", "CreateAjax", ex, "No se pudo crear el registro");
            }
        }


        public string SubirAdjunto(HttpPostedFileBase Archivo, string Tipo, string Id_Foto)
        {
            string enlace;

            if (ValidarArchivo(Archivo))
            {
                int Ano = Convert.ToInt32(DateTime.Now.Year.ToString());
                string Month = DateTime.Now.Month.ToString();
                string Day = DateTime.Now.Day.ToString();
                string Hora = DateTime.Now.Hour.ToString();
                string Minutos = DateTime.Now.Minute.ToString();
                string Segundos = DateTime.Now.Second.ToString();
                string Ruta = System.Configuration.ConfigurationManager.AppSettings["AdjuntosPMU"];
                string Ubicacion = "IRISP1/EXPENDIOSWEB";
                string filename = Id_Foto + Tipo;
                string Directorio = Ruta + Ubicacion + "/" + Ano + "/" + "/" + Month;

                //valida si existe el directorio de lo contrario lo crea
                if (!Directory.Exists(Directorio))
                    Directory.CreateDirectory(Directorio);
                enlace = Directorio + "/" + filename;
                Archivo.SaveAs(enlace);
                //Actualizar ruta
                return "/" + Ano + "/" + "/" + Month + "/" + filename;
            }
            else
            {
                return "0";
            }
        }

        public bool ValidarArchivo(HttpPostedFileBase _archivo)
        {
            //1. El archivo no puede ser nulo
            if (_archivo == null)
                return false;

            ////2. Solo se permite extensión jpg
            //if (_archivo.ContentType != "application/png")
            //    return false;

            //3. el tamaño permito solo es 24 MB
            if (_archivo.ContentLength >= 909715100)
                return false;
            return true;
        }

        // Método auxiliar para guardar el archivo en el servidor o almacenamiento
        private string GuardarArchivoEnElServidor(IFormFile archivo)
        {
            // Lógica para guardar el archivo en el servidor
            // Puedes utilizar System.IO para guardar en una carpeta del servidor
            // o servicios de almacenamiento como Azure Blob Storage.

            // Ejemplo para guardar en una carpeta del servidor:
            var rutaDeGuardado = Path.Combine("CarpetaDeFotos", archivo.FileName);
            using (var stream = new FileStream(rutaDeGuardado, FileMode.Create))
            {
                archivo.CopyTo(stream);
            }

            return rutaDeGuardado;
        }



        [HttpPost]
        [Authorize(Roles = ("Administrador,Administrador IRISP1"))]
        public async Task<ActionResult> ActualizarAjax(CriminalidadDTO criminalidadDTO)
        {
            try
            {
                if (criminalidadDTO.ID_CLASE == null || criminalidadDTO.NOMBRE_CLASE == null || criminalidadDTO.CANTIDAD_INTEGRANTE == null ||
                    criminalidadDTO.ID_FUENTE == null || criminalidadDTO.CARACTERISTICAS_GENERALES == null)
                    return Json(new { ok = false, mensaje = "Faltan campos de información para actualizar el IRISP1 !!!" });

                criminalidadDTO.IDENTIFICACION_MODIFICA = Convert.ToInt32(gestionClaims.Identificacion);
                criminalidadDTO.MAQUINA_MODIFICA = Request.UserHostAddress;

                if (ModelState.IsValid)
                    if (await new CriminalidadRepositorio().Actualizar(criminalidadDTO))
                        return Json(new { ok = true, mensaje = "Se actualizó correctamente el registro" });

                string listaErrores = null;

                foreach (var key in ModelState)
                {
                    if (key.Value.Errors.Count > 0)
                        listaErrores += "<li>" + key.Value.Errors[0].ErrorMessage + "</li>";
                }

                if (string.IsNullOrEmpty(listaErrores))
                    return Json(new { ok = false, mensaje = "No se pudo actualizar el registro" });
                else
                    return Json(new { ok = false, mensaje = "<ul>" + listaErrores + "</ul>" });
            }
            catch (Exception ex)
            {
                return await CapturaError("POST", "Irisp", "Criminalidad", "ActualizarAjax", ex, "No se pudo actualizar el registro");
            }
        }

        public async Task<ActionResult> ConsultarCriminalidadEtapaVerificacion(int? variables)
        {
            try
            {
                ParametrosPaginacionDTO parametrosPaginacionDTO = new ParametrosPaginacionDTO
                {
                    Start = Convert.ToInt32(Request["start"]),
                    Length = Convert.ToInt32(Request["length"]),
                    SearchValue = Request["search[value]"],
                    SortColumnName = Request["columns[" + Request["order[0][column]"] + "][name]"],
                    SortDirection = Request["order[0][dir]"]
                };

                CriminalidadRepositorio core = new CriminalidadRepositorio();
                List<CriminalidadDTO> resultadosConsulta = new List<CriminalidadDTO>();

                if (User.IsInRole("Administrador") || User.IsInRole("Administrador IRISP1"))
                    resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacion(parametrosPaginacionDTO, variables);
                else if (User.IsInRole("Regional IRISP1"))
                    resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacionPorRegion(parametrosPaginacionDTO, gestionClaims.CodigoRegional, variables);
                else if (User.IsInRole("Funcional Unidad IRISP1") || User.IsInRole("Seguimiento IRISP1")) { 

                    if (gestionClaims.UnidadFisica == "DESAB")  { 
                        resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacionPorUnidad(parametrosPaginacionDTO, "REMSA", variables); 
                    } else
                    {
                        resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacionPorUnidad(parametrosPaginacionDTO, gestionClaims.UnidadFisica, variables);
                    }
                }

                else if (User.IsInRole("Funcional Dependencia IRISP1"))
                {

                    if (gestionClaims.UnidadFisica == "DESAB")
                    {
                        decimal codigoAnterior = (decimal)await new UnidadesRepositorio().ConsultarUnidadAnterior(gestionClaims.CodigoDependencia);

                        resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacionPorDependencia(parametrosPaginacionDTO, (int)codigoAnterior, variables);

                        if (resultadosConsulta.Count == 0)
                        {
                            decimal? idUnidad = await new UnidadesRepositorio().ConsultarUnidadFuncionario(codigoAnterior);
                            if (idUnidad != null)
                                resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacionPorDependencia(parametrosPaginacionDTO, (int)idUnidad, variables);
                        }

                    }
                    else
                    {
                        resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacionPorDependencia(parametrosPaginacionDTO, (int)gestionClaims.CodigoDependencia, variables);

                        if (resultadosConsulta.Count == 0)
                        {
                            decimal? idUnidad = await new UnidadesRepositorio().ConsultarUnidadFuncionario(gestionClaims.CodigoDependencia);
                            if (idUnidad != null)
                                resultadosConsulta = await core.ConsultarRegistrosEtapaVerificacionPorDependencia(parametrosPaginacionDTO, (int)idUnidad, variables);
                        }
                    }


                }

                return Json(new { data = resultadosConsulta, draw = Request["draw"], recordsTotal = parametrosPaginacionDTO.CantidadRegistrosTotal, recordsFiltered = parametrosPaginacionDTO.CantidadRegistrosFiltrados });
            }
            catch (Exception ex)
            {
                return await CapturaError("POST", "Irisp", "Criminalidad", "ConsultarCriminalidadEtapaVerificacion", ex, "No se pudo consultar y enviar el JSON de los IRISP1");
            }
        }

                    resultadosConsulta = await core.ConsultarRegistrosEtapaFinalPorRegion(parametrosPaginacionDTO, gestionClaims.CodigoRegional, variables);
                else if (User.IsInRole("Funcional Unidad IRISP1") || User.IsInRole("Seguimiento IRISP1"))
                {
                    if (gestionClaims.UnidadFisica == "DESAB")
                    {

                      
                        resultadosConsulta = await core.ConsultarRegistrosEtapaFinalPorUnidad(parametrosPaginacionDTO, "REMSA", variables);
                    }
                    else
                    {
                        resultadosConsulta = await core.ConsultarRegistrosEtapaFinalPorUnidad(parametrosPaginacionDTO, gestionClaims.UnidadFisica, variables);
                    }
                }

               
                else if (User.IsInRole("Funcional Dependencia IRISP1"))
                {

                    if (gestionClaims.UnidadFisica == "DESAB")
                    {
                        decimal codigoAnterior = (decimal)await new UnidadesRepositorio().ConsultarUnidadAnterior(gestionClaims.CodigoDependencia);
                        resultadosConsulta = await core.ConsultarRegistrosEtapaFinalPorDependencia(parametrosPaginacionDTO, (int)codigoAnterior, variables);

                        if (resultadosConsulta.Count == 0)
                        {
                            decimal? idUnidad = await new UnidadesRepositorio().ConsultarUnidadFuncionario(codigoAnterior);
                            if (idUnidad != null)
                                resultadosConsulta = await core.ConsultarRegistrosEtapaFinalPorDependencia(parametrosPaginacionDTO, (int)idUnidad, variables);
                        }

                    }
                    else
                    {
                        resultadosConsulta = await core.ConsultarRegistrosEtapaFinalPorDependencia(parametrosPaginacionDTO, (int)gestionClaims.CodigoDependencia, variables);

                        if (resultadosConsulta.Count == 0)
                        {
                            decimal? idUnidad = await new UnidadesRepositorio().ConsultarUnidadFuncionario(gestionClaims.CodigoDependencia);
                            if (idUnidad != null)
                                resultadosConsulta = await core.ConsultarRegistrosEtapaFinalPorDependencia(parametrosPaginacionDTO, (int)idUnidad, variables);
                        }
                    }


                   
                }

                return Json(new { data = resultadosConsulta, draw = Request["draw"], recordsTotal = parametrosPaginacionDTO.CantidadRegistrosTotal, recordsFiltered = parametrosPaginacionDTO.CantidadRegistrosFiltrados });
            }
            catch (Exception ex)

        public async Task<FileResult> DescargarIrisp1(int? anio)
        {
            string ruta = System.Web.Configuration.WebConfigurationManager.AppSettings["servidorArchivosExcel"];
            string pathFile = ruta + $"{DateTime.Now.ToString("yyyyMMddHHmmss")}.xlsx";
            System.Data.DataTable dt = new System.Data.DataTable();

            //Columnas
            dt.Columns.Add("Estado", typeof(string));
            dt.Columns.Add("Estado existencia", typeof(string));
            dt.Columns.Add("Código", typeof(string));
            dt.Columns.Add("Región", typeof(string));
            dt.Columns.Add("Unidad", typeof(string));
            dt.Columns.Add("Dependencia", typeof(string));
            dt.Columns.Add("Municipio", typeof(string));
            dt.Columns.Add("Fecha Inicio Actividad", typeof(string));
            dt.Columns.Add("Clase", typeof(string));
            dt.Columns.Add("Nombre clase", typeof(string));
            dt.Columns.Add("Cantidad", typeof(string));
            dt.Columns.Add("Caracteristicas generales", typeof(string));
            dt.Columns.Add("Descripción tramite", typeof(string));
            dt.Columns.Add("Zona", typeof(string));
            dt.Columns.Add("Tipo servicio", typeof(string));
            dt.Columns.Add("Fuente", typeof(string));
            dt.Columns.Add("Fecha creacion", typeof(string));
            dt.Columns.Add("Unidad Verificación Existencia", typeof(string));
            dt.Columns.Add("Fecha Asignación Verificación Existencia", typeof(string));
            dt.Columns.Add("Fecha Respuesta Verificación Existencia", typeof(string));
            dt.Columns.Add("Contador Verificación Existencia", typeof(string));
            dt.Columns.Add("Unidad Proceso Investigativo", typeof(string));
            dt.Columns.Add("Fecha Asignación Proceso Investigativo", typeof(string));
            dt.Columns.Add("Fecha Respuesta Proceso Investigativo", typeof(string));
            dt.Columns.Add("Contador Proceso Investigativo", typeof(string));
            dt.Columns.Add("Resultados", typeof(string));

            //Filas
            Stopwatch sw = new Stopwatch();
            sw.Start();
            CriminalidadRepositorio core = new CriminalidadRepositorio();
            List<CriminalidadDTO> resultadosConsulta = new List<CriminalidadDTO>();

            if (User.IsInRole("Administrador") || User.IsInRole("Administrador IRISP1") || User.IsInRole("Consulta IRISP1"))
                resultadosConsulta = await core.ConsultarRegistrosReporte(anio);
            else if (User.IsInRole("Regional IRISP1"))
                resultadosConsulta = await core.ConsultarRegistrosPorRegionReporte(gestionClaims.CodigoRegional, anio);
            else if (User.IsInRole("Funcional Unidad IRISP1") || User.IsInRole("Seguimiento IRISP1"))
                resultadosConsulta = await core.ConsultarRegistrosPorUnidadReporte(gestionClaims.UnidadFisica, anio);
            else if (User.IsInRole("Funcional Dependencia IRISP1"))
            {
                resultadosConsulta = await core.ConsultarRegistrosPorDependenciaReporte((int)gestionClaims.CodigoDependencia, anio);

                if (resultadosConsulta.Count == 0)
                {
                    decimal? idUnidad = await new UnidadesRepositorio().ConsultarUnidadFuncionario((decimal)gestionClaims.CodigoDependencia);
                    if (idUnidad != null)
                        resultadosConsulta = await core.ConsultarRegistrosPorDependenciaReporte((int)idUnidad, anio);
                }
            }

            foreach (var item in resultadosConsulta)
            {
                dt.Rows.Add(
                    item.ID_ESTADO_STR,
                    item.ID_ESTADO_EXISTENCIA_STR,
                    item.CODIGO,
                    item.REGION,
                    item.SIGLA_PAPA,
                    item.ID_UNIDAD_STR,
                    item.MUNICIPIO,
                    item.FECHA_INICIO_EXISTENCIA_STR,
                    item.ID_CLASE_STR,
                    item.NOMBRE_CLASE,
                    item.CANTIDAD_INTEGRANTE,
                    item.CARACTERISTICAS_GENERALES,
                    item.DESCRIPCION_TRAMITE,
                    item.ID_ZONA_STR,
                    item.ID_TIPO_SERVICIO_STR,
                    item.ID_FUENTE_STR,
                    item.FECHA_CREACION,
                    item.UNIDAD_VERIFICACION,
                    item.FECHA_ASIGNACION_VERIFICA,
                    item.FECHA_RESPUESTA_VERIFICA,
                    item.CONTADOR_VERIFICA,
                    item.UNIDAD_PROCESO_INVESTIGATIVO,
                    item.FECHA_ASIGNACION_INVESTIGA,
                    item.FECHA_RESPUESTA_INVESTIGA,
                    item.CONTADOR_INVESTIGA,
                    item.IRISP_RESULTADO_STR
                    );
            }

            SLDocument sLDocument = new SLDocument();
            sLDocument.ApplyNamedCellStyleToRow(1, SLNamedCellStyleValues.Neutral);
            sLDocument.ImportDataTable(1, 1, dt, true);
            sLDocument.SaveAs(pathFile);
            sw.Stop();
            new UtilidadesRepositorio().EliminarArchivo(pathFile, Convert.ToInt32(sw.ElapsedMilliseconds));
            decimal identificacion = Convert.ToDecimal(gestionClaims.Identificacion);
            new AdministracionUsuarios().RegistroAuditoria(new AuditoriaDTO { ID_PROCESO = (int)EnumListaAcciones.DescargaReporte, USUARIO = gestionClaims.UsuarioEmpresarial, IDENTIFICACION = identificacion, DESCRIPCION = "Descarga Reporte IRISP1", MAQUINA = Request.UserHostAddress });
            return File(pathFile, "application/xlsx", $"ReporteIrisp.xlsx");
        }

                    Ext = ".jpg";

                else if (bmpReturn.RawFormat.Equals(ImageFormat.Png))

                    ContenType = "image/png";

        {
            try
            {

                Datos.ID_FOTOS_CRIMINALIDAD = Guid.NewGuid().ToString();
                Datos.tipo_doc = Path.GetExtension(Datos.FOTO_CRIMINALIDAD.FileName);
                Datos.ruta = SubirAdjunto(Datos.FOTO_CRIMINALIDAD, Datos.tipo_doc, Datos.ID_FOTOS_CRIMINALIDAD);

                        return Json(new { ok = true, mensaje = "Se agregó correctamente el registro" });

                string listaErrores = null;
                int contadorErrores = 0;

                foreach (var key in ModelState)
                {
                    if (key.Value.Errors.Count > 0)
                    {
                        contadorErrores++;
                        listaErrores += $" {contadorErrores}. {key.Value.Errors[0].ErrorMessage}";
                    }
                }

                if (string.IsNullOrEmpty(listaErrores))
                    return Json(new { ok = false, mensaje = "No se pudo guardar el registro" });
                else
                    return Json(new { ok = false, mensaje = listaErrores });
            }
            catch (Exception ex)
            {
                return await CapturaError("POST", "Irisp", "Criminalidad", "CargarFoto", ex, "No se pudo cargar la foto");
            }
        }

        [HttpPost]
        public async Task<ActionResult> CreateAjax(CriminalidadDirecDTO criminalidadDirecDTO, List<IntegranteDirecDTO> lstIntegranteDirecDTO, List<DelitoDirecDTO> lstDelitoDirecDTO)
        {
            try
            {
                if (lstIntegranteDirecDTO != null)
                {
                    foreach (var integranteDirecDTO in lstIntegranteDirecDTO)
                    {
                        if (integranteDirecDTO.ALIAS == null && integranteDirecDTO.APELLIDO == null && integranteDirecDTO.IDENTIFICACION == null && integranteDirecDTO.NOMBRE == null)
                            return Json(new { ok = false, mensaje = "Tiene que registrar por lo menos algún campo del integrante !!!" });
                        else if (integranteDirecDTO.IDENTIFICACION != null && (integranteDirecDTO.APELLIDO == null && integranteDirecDTO.ALIAS == null && integranteDirecDTO.NOMBRE == null))
                            return Json(new { ok = false, mensaje = "No puede registrar soló la cedula, digite en más campos !!!" });
                    }
                }

                if (lstDelitoDirecDTO.Count == 0)
                    return Json(new { ok = false, mensaje = "Seleccione al menos un delito !!!" });

                if (string.IsNullOrEmpty(criminalidadDirecDTO.LONGITUD) && string.IsNullOrEmpty(criminalidadDirecDTO.LATITUD))
                    return Json(new { ok = false, mensaje = "La latitud y longitud no pueden estar vacias, seleccione un punto geográfico valido con coordenadas" });

                if (criminalidadDirecDTO.MUNICIPIO == null)
                    return Json(new { ok = false, mensaje = "La ubicación debe ser dentro del territorio Colombiano !!!" });

                criminalidadDirecDTO.IDENTIFICACION_CREACION = Convert.ToInt32(gestionClaims.Identificacion);
                criminalidadDirecDTO.MAQUINA_CREACION = Request.UserHostAddress;

                if (ModelState.IsValid)
                    if (await new CriminalidadDirecRepositorio().Agregar(criminalidadDirecDTO, lstIntegranteDirecDTO, lstDelitoDirecDTO))
                        return Json(new { ok = true, mensaje = "Se agregó correctamente el registro" });

                string listaErrores = null;
                int contadorErrores = 0;

                foreach (var key in ModelState)
                {
                    if (key.Value.Errors.Count > 0)
                    {
                        contadorErrores++;
                        listaErrores += $" {contadorErrores}. {key.Value.Errors[0].ErrorMessage}";
                    }
                }

                if (string.IsNullOrEmpty(listaErrores))
                    return Json(new { ok = false, mensaje = "No se pudo guardar el registro" });
                else
                    return Json(new { ok = false, mensaje = listaErrores });
            }
            catch (Exception ex)
            {
                return await CapturaError("POST", "Irisp", "CriminalidadDirec", "CreateAjax", ex, "No se pudo crear el registro");
            }
        }

        [HttpPost]
        public async Task<ActionResult> ActualizarAjax(CriminalidadDirecDTO criminalidadDirecDTO)
        {
            try
            {
                if (criminalidadDirecDTO.ID_ESTADO == (int)EnumeracionEstadoIrispDirecciones.Investigacion)
                {
                    if (string.IsNullOrEmpty(criminalidadDirecDTO.NUNC))
                        return Json(new { ok = false, mensaje = "El NUNC es obligatorio !!!" });

                    //if (string.IsNullOrEmpty(criminalidadDirecDTO.CODIGO_MORED) || string.IsNullOrEmpty(criminalidadDirecDTO.NOMBRE_MORED))
                    //    return Json(new { ok = false, mensaje = "El código y nombre de la MORED es obligatorio" });
                }
                else if (criminalidadDirecDTO.ID_ESTADO == (int)EnumeracionEstadoIrispDirecciones.Finalizado)
                {
                    if (criminalidadDirecDTO.ERRADICADO == null)
                        return Json(new { ok = false, mensaje = "El campo Erradicado debe estar diligenciado !!!" });
                    
                    if (criminalidadDirecDTO.ERRADICADO.Value)
                        if (string.IsNullOrWhiteSpace(criminalidadDirecDTO.SIEDCO))
                            return Json(new { ok = false, mensaje = "El campo SIEDCO debe estar diligenciado !!!" });
                }
                else if (criminalidadDirecDTO.ID_ESTADO == (int)EnumeracionEstadoIrispDirecciones.Descartado)
                {
                    if (string.IsNullOrEmpty(criminalidadDirecDTO.OBSERVACION))
                        return Json(new { ok = false, mensaje = "El campo Observación debe estar diligenciado !!!" });
                }

                criminalidadDirecDTO.IDENTIFICACION_MODIFICA = Convert.ToInt32(gestionClaims.Identificacion);
                criminalidadDirecDTO.MAQUINA_MODIFICA = Request.UserHostAddress;

                if (ModelState.IsValid)
                    if (await new CriminalidadDirecRepositorio().ActualizarEstado(criminalidadDirecDTO))
                        return Json(new { ok = true, mensaje = "Se actualizó correctamente el registro" });

                string listaErrores = null;
                int contadorErrores = 0;

                foreach (var key in ModelState)
                {
                    if (key.Value.Errors.Count > 0)
                    {
                        contadorErrores++;
                        listaErrores += $" {contadorErrores}. {key.Value.Errors[0].ErrorMessage}";
                    }
                }

                if (string.IsNullOrEmpty(listaErrores))
                    return Json(new { ok = false, mensaje = "No se pudo guardar el registro" });
                else
                    return Json(new { ok = false, mensaje = listaErrores });
            }
            catch (Exception ex)
            {
                return await CapturaError("POST", "Irisp", "CriminalidadDirec", "ActualizarAjax", ex, "No se pudo actualizar el registro");
            }
        }

        [HttpPost]
        public async Task<ActionResult> ConsultaGeneral()
        {
            try
            {
                object resultadoBusqueda = null;

                if (User.IsInRole("Administrador") || User.IsInRole("Administrador IRISP1") || User.IsInRole("Consulta IRISP1"))
                {
                    resultadoBusqueda = await new CriminalidadDirecRepositorio().ConsultaGeneral();
                }
                else if (User.IsInRole("Regional Investigacion Criminal"))
                {
                    resultadoBusqueda = await new CriminalidadDirecRepositorio().ConsultaRegion((int)gestionClaims.CodigoRegional);
                }
                else if (User.IsInRole("Verificacion IRISP1"))
                {
                    if (gestionClaims.UnidadFisica == "DESAB")
                    {
                        resultadoBusqueda = await new CriminalidadDirecRepositorio().ConsultaUnidad("REMSA");
                    }
                    else
                    {
                        resultadoBusqueda = await new CriminalidadDirecRepositorio().ConsultaUnidad(gestionClaims.UnidadFisica);
                    }
                }

                // IMPORTANTE: Configurar el JSON aquí para TODOS los casos
                var jsonResult = Json(new { data = resultadoBusqueda }, JsonRequestBehavior.AllowGet);
                jsonResult.MaxJsonLength = int.MaxValue;
                return jsonResult;
            }
            catch (Exception ex)
            {
                // Esto te ayudará a ver el error real en la consola del navegador
                Response.StatusCode = 500;
                return Json(new { error = ex.Message, stack = ex.StackTrace });
            }
        }

            dt.Columns.Add("Longitud", typeof(string));
            dt.Columns.Add("Cuadrante", typeof(string));
            dt.Columns.Add("Municipio", typeof(string));

            dt.Columns.Add("Total Capturas", typeof(string));
            dt.Columns.Add("Total Allanamientos", typeof(string));
            dt.Columns.Add("Total Incaptaciones", typeof(string));

            //Filas
            Stopwatch sw = new Stopwatch();
            sw.Start();
            CriminalidadDirecRepositorio core = new CriminalidadDirecRepositorio();
            List<CriminalidadDirecDTO> resultadosConsulta = new List<CriminalidadDirecDTO>();

            if (User.IsInRole("Administrador") || User.IsInRole("Administrador IRISP1") || User.IsInRole("Consulta IRISP1"))
                resultadosConsulta = await core.ConsultaGeneral();
            else if (User.IsInRole("Regional Investigacion Criminal"))
                resultadosConsulta = await core.ConsultaRegion((int)gestionClaims.CodigoRegional);
            else if (User.IsInRole("Verificacion IRISP1"))
                resultadosConsulta = await core.ConsultaUnidad(gestionClaims.UnidadFisica);


            foreach (var item in resultadosConsulta)
            {
                dt.Rows.Add(
                    item.CODIGO,
                    item.UNIDAD_INFORMA_STR,
                    item.SIGLA_UNIDAD_INFORMA_STR,
                    item.REGION,
                    item.UNIDAD_STR,
                    item.SIGLA_UNIDAD_STR,

                    item.ID_FUENTE_STR,

                    item.CARACTERISTICAS_GENERALES,
                    item.ID_CATEGORIA_STR,
                    item.OTRA_CATEGORIA,
                    item.CODIGO_MORED,
                    item.NOMBRE_MORED,
                    item.NUNC,
                    item.SIEDCO,
                    item.ERRADICADO,
                    item.BARRIO,
                    item.DIRECCION,
                    item.LATITUD,
                    item.LONGITUD,
                    item.CUADRANTE,
                    item.MUNICIPIO,
                    item.IRISP_RESULTADO_DIREC.Where(y => y.ID_TIPO== 79).Sum(x => x.CANTIDAD),
                    item.IRISP_RESULTADO_DIREC.Where(y => y.ID_TIPO == 78).Sum(x => x.CANTIDAD),
                    item.IRISP_RESULTADO_DIREC.Where(y => y.ID_TIPO == 77).Sum(x => x.CANTIDAD)
                    );
            }

            SLDocument sLDocument = new SLDocument();
            sLDocument.ApplyNamedCellStyleToRow(1, SLNamedCellStyleValues.Neutral);
            sLDocument.ImportDataTable(1, 1, dt, true);
            sLDocument.SaveAs(pathFile);
            sw.Stop();
            new UtilidadesRepositorio().EliminarArchivo(pathFile, Convert.ToInt32(sw.ElapsedMilliseconds));
            decimal identificacion = Convert.ToDecimal(gestionClaims.Identificacion);
            new AdministracionUsuarios().RegistroAuditoria(new AuditoriaDTO { ID_PROCESO = (int)EnumListaAcciones.DescargaReporte, USUARIO = gestionClaims.UsuarioEmpresarial, IDENTIFICACION = identificacion, DESCRIPCION = "Descarga Reporte Expendios", MAQUINA = Request.UserHostAddress });

        }

                if (_id == null)

                var resultadoTarea = await new TareaRepositorio().ConsultarRegistrosPorId(_id);
                ViewBag.TAREA = resultadoTarea;


                foreach (var item in resultadoTarea)
                {
                    if (item.ID_LISTA_TAREAS == (int)EnumeracionTarea.Asignacion_Proceso_Investigativo)

                        ViewBag.RESULTADO = await new ResultadoRepositorio().ConsultarRegistrosPorId(_idCriminalidad);

                    {
                        ViewBag.RESULTADO = null;

                // Lista para modal de creacion Resultado
                ViewBag.ID_TIPO = new SelectList(await new ListasRepositorio().ListaDominio((int)EnumeracionListasDominios.Lista_Tipo_Resultado), "DOMINIO_ID", "DESCRIPCION");

                ViewBag.CRIMINALIDAD_ID = _idCriminalidad;
                return View();
            }
            catch (Exception ex)
            {
                return await CapturaError("GET", "Irisp", "ResponsabilidadValidacion", "Details", ex, "No se pudo ingresar al formulario");

        public async Task<ActionResult> ConsultarCriminalidadEtapaVerificacion(int? variables)
        {
            try
            {
                ParametrosPaginacionDTO parametrosPaginacionDTO = new ParametrosPaginacionDTO
                {
                    Start = Convert.ToInt32(Request["start"]),
                    Length = Convert.ToInt32(Request["length"]),
                    SearchValue = Request["search[value]"],
                    SortColumnName = Request["columns[" + Request["order[0][column]"] + "][name]"],
                    SortDirection = Request["order[0][dir]"]
                };

                CriminalidadRepositorio core = new CriminalidadRepositorio();
                List<CriminalidadDTO> resultadosConsulta = new List<CriminalidadDTO>();

                if (User.IsInRole("Administrador") || User.IsInRole("Administrador IRISP1") || User.IsInRole("Admin PreAisec"))
                    resultadosConsulta = await new ResponsableValidacionRepositorio().ConsultarRegistrosCriminalidadEtapaVerificacion(parametrosPaginacionDTO, variables);
                else if (User.IsInRole("Verificacion IRISP1"))
                {

                    if (gestionClaims.UnidadFisica == "DESAB")
                    {
                        // 1. Captura como nullable
                        decimal codigoAnterior = 0;
                        decimal? codigoAnteriorNullable = await new UnidadesRepositorio().ConsultarUnidadAnterior(gestionClaims.CodigoDependencia);

                        // 2. Valida antes de usar
                        if (codigoAnteriorNullable.HasValue)
                        {
                             codigoAnterior = codigoAnteriorNullable.Value;
                            resultadosConsulta = await new ResponsableValidacionRepositorio()
                                .ConsultarRegistrosCriminalidadEtapaVerificacionPorDependencia(parametrosPaginacionDTO, codigoAnterior, variables);
                        }
                        else
                        {
                            // Manejo en caso de que no exista unidad anterior (ej. lista vacía o lógica alternativa)
                            resultadosConsulta = new List<CriminalidadDTO>();
                        }

                        resultadosConsulta = await new ResponsableValidacionRepositorio().ConsultarRegistrosCriminalidadEtapaVerificacionPorDependencia(parametrosPaginacionDTO, (decimal)codigoAnterior, variables);

                        if (resultadosConsulta.Count == 0)
                        {
                            decimal? idUnidad = await new UnidadesRepositorio().ConsultarUnidadFuncionario((decimal)codigoAnterior);
                            if (idUnidad != null)
                            {
                                resultadosConsulta = await new ResponsableValidacionRepositorio().ConsultarRegistrosCriminalidadEtapaVerificacionPorDependencia(parametrosPaginacionDTO, idUnidad, variables);
                                if (resultadosConsulta.Count == 0)
                                {
                                    decimal? idUnidad2 = await new UnidadesRepositorio().ConsultarUnidadFuncionario((decimal)idUnidad);
                                    if (idUnidad2 != null)
                                        resultadosConsulta = await new ResponsableValidacionRepositorio().ConsultarRegistrosCriminalidadEtapaVerificacionPorDependencia(parametrosPaginacionDTO, idUnidad2, variables);
                                }
                            }
                        }

                    }
                    else
                    {
                        resultadosConsulta = await new ResponsableValidacionRepositorio().ConsultarRegistrosCriminalidadEtapaVerificacionPorDependencia(parametrosPaginacionDTO, (decimal)gestionClaims.CodigoDependencia, variables);

                        if (resultadosConsulta.Count == 0)
                        {
                            decimal? idUnidad = await new UnidadesRepositorio().ConsultarUnidadFuncionario((decimal)gestionClaims.CodigoDependencia);
                            if (idUnidad != null)
                            {
                                resultadosConsulta = await new ResponsableValidacionRepositorio().ConsultarRegistrosCriminalidadEtapaVerificacionPorDependencia(parametrosPaginacionDTO, idUnidad, variables);
                                if (resultadosConsulta.Count == 0)

                                    decimal? idUnidad2 = await new UnidadesRepositorio().ConsultarUnidadFuncionario((decimal)idUnidad);
                                    if (idUnidad2 != null)
                                        resultadosConsulta = await new ResponsableValidacionRepositorio().ConsultarRegistrosCriminalidadEtapaVerificacionPorDependencia(parametrosPaginacionDTO, idUnidad2, variables);
                                }
                            }
                        }
                    }

                }

                return Json(new { data = resultadosConsulta, draw = Request["draw"], recordsTotal = parametrosPaginacionDTO.CantidadRegistrosTotal, recordsFiltered = parametrosPaginacionDTO.CantidadRegistrosFiltrados });

        [HttpPost]
        public async Task<ActionResult> CreateAjax(ResponsabilidadValidacionDTO responsabilidadValidacionDTO, TareaDTO tareaDTO)
        {
            try
            {
                if (responsabilidadValidacionDTO.CRIMINALIDAD_ID == null || responsabilidadValidacionDTO == null || tareaDTO == null)
                    return Json(new { ok = false, mensaje = "No se encuentra un identificador valido para asociar el delito" });

                if (responsabilidadValidacionDTO.ID_UNIDAD_RESPON == 0 || responsabilidadValidacionDTO.ID_UNIDAD_RESPON == null)
                    return Json(new { ok = false, mensaje = "Seleccione una dependencia" });

                if (tareaDTO.ID_LISTA_TAREAS == null)
                    return Json(new { ok = false, mensaje = "Seleccione una tarea a asignar" });

                if (tareaDTO.ID_LISTA_TAREAS == (int)EnumeracionTarea.Verificacion_Existencia)
                    if (await new ResponsableValidacionRepositorio().ExistenciaResponsableVerificacion(responsabilidadValidacionDTO.CRIMINALIDAD_ID))
                        return Json(new { ok = false, mensaje = "Ya se registró una unidad responsable de la verificación de la existencia ó existen tareas de verificación no rechazadas" });

                if (tareaDTO.ID_LISTA_TAREAS == (int)EnumeracionTarea.Asignacion_Proceso_Investigativo)
                    if (!await new CriminalidadRepositorio().ConsultaExistenciaIRISP(responsabilidadValidacionDTO.CRIMINALIDAD_ID))
                        return Json(new { ok = false, mensaje = "Para asignar el proceso investigativo, el IRISP debe estar en estado existencia 'Si Existe' " });
                    else if (await new ResponsableValidacionRepositorio().ExistenciaResponsableInvestigacion(responsabilidadValidacionDTO.CRIMINALIDAD_ID))
                        return Json(new { ok = false, mensaje = "Ya se registró una unidad responsable del proceso investigativo ó existen tareas de investigación no rechazadas" });

                responsabilidadValidacionDTO.IDENTIFICACION_CREACION = Convert.ToInt32(gestionClaims.Identificacion);
                responsabilidadValidacionDTO.MAQUINA_CREACION = Request.UserHostAddress;

                if (ModelState.IsValid)
                {
                    if (await new ResponsableValidacionRepositorio().Agregar(responsabilidadValidacionDTO, tareaDTO))
                        if (true)
                        {
                            if (tareaDTO.ID_LISTA_TAREAS == (int)EnumeracionTarea.Verificacion_Existencia)
                            {
                                Criminalidad_DTO criminalidadDTO = await new CriminalidadRepositorio().ConsultarRegistroPorId(responsabilidadValidacionDTO.CRIMINALIDAD_ID);
                                var ExitoCorreo = new CorreoRepositorio().EnvioCorreoBasico(Convert.ToInt64(criminalidadDTO.IDENTIFICACION_INFORMA), "Notificación IRISP1.", Convert.ToInt64(criminalidadDTO.IDENTIFICACION_INFORMA), "Se realizó cambio de estado al caso IRISP1: " + criminalidadDTO.CODIGO + " a VERIFICACIÓN DE LA EXISTENCIA (ASIGNADO), para ver detalles de la modificación, ingrese al sistema de información IRISP1.");
                                return Json(new { ok = true, mensaje = "Se agregó correctamente el registro" });
                            }else if (tareaDTO.ID_LISTA_TAREAS == (int)EnumeracionTarea.Asignacion_Proceso_Investigativo)
                            {

                }
            }

            lista = Json(new { data = resp });
            lista.MaxJsonLength = int.MaxValue;
            return lista;
        }

    }
