-- ============================================================================
-- PK_CONTROL_GESTION_IRIS
--
-- Módulo de Control de Gestión (tiempos/SLA) de IRIS-P1.
-- Paquete NUEVO: requiere ejecutar primero Oracle/IRISP_SLA_CONFIG_DDL.txt.
--
-- Convenciones seguidas (iguales al resto del sistema, ver PK_VERIFICACION_IRIS,
-- PK_SEGUIMIENTO_IRIS, PK_INTEGRANTES_IRIS):
--   - P_RESULTADO OUT NUMBER + SRV_Message OUT VARCHAR2 para operaciones de escritura.
--   - Borrado lógico (VIGENTE=0), nunca DELETE físico.
--   - Errores registrados vía Pks_Administracion_IRIS.P_InsErrorsystem.
--   - Filtro de roles idéntico al de PK_VERIFICACION_IRIS.F_GetInfoGrillas
--     (roles 1,2 ven todo; rol 8 ve solo lo de su unidad responsable), para no
--     sumar un modelo de visibilidad más a los ya existentes.
-- ============================================================================

CREATE OR REPLACE PACKAGE PK_CONTROL_GESTION_IRIS AS

  -- Catálogo de SLA configurados (uno vigente por tipo de tarea, dominio PADRE_ID=51)
  FUNCTION F_GetSlaConfig RETURN SYS_REFCURSOR;

  -- Upsert por ID_LISTA_TAREAS: si ya existe un SLA vigente para ese tipo de tarea, lo actualiza;
  -- si no, lo crea. Valida que P_TiempoAlertaHoras < P_TiempoMaximoHoras.
  PROCEDURE P_InsUpdSlaConfig(
    P_IdListaTareas      IN NUMBER,
    P_TiempoAlertaHoras  IN NUMBER,
    P_TiempoMaximoHoras  IN NUMBER,
    P_Justificacion      IN VARCHAR2,
    P_Usuario            IN NUMBER,
    P_Maquina            IN VARCHAR2,
    P_RESULTADO          OUT NUMBER,
    SRV_Message          OUT VARCHAR2
  );

  -- Borrado lógico de una configuración de SLA
  PROCEDURE P_DelSlaConfig(
    P_SlaConfigId  IN VARCHAR2,
    P_Usuario      IN NUMBER,
    P_Maquina      IN VARCHAR2,
    P_RESULTADO    OUT NUMBER,
    SRV_Message    OUT VARCHAR2
  );

  -- Grilla principal del tablero: una fila por tarea vigente en el rango de fechas, con las
  -- horas transcurridas y el estado de SLA ya calculados. Filtrable también por región y por
  -- unidad (SIGLA_UNIDAD), igual que el mapa de puntos, para evaluar una unidad/región en un
  -- periodo concreto. P_RegionCodigo es informativo/redundante cuando P_SiglaUnidad ya viene
  -- informado (la unidad ya determina la región), pero permite filtrar por toda una región sin
  -- elegir una unidad puntual.
  FUNCTION F_GetTareasControlGestion(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR;

  -- KPIs de tiempo de gestión a NIVEL DE CASO (no de tarea): una fila por IRISP1 del rango de
  -- fechas, con el tiempo total de creación a finalización y el tiempo por etapa
  -- (Verificación/Investigación). Calculado de forma set-based (ROW_NUMBER analítico) en vez de
  -- invocar por fila las funciones escalares F_Fecha_Verif_Existe/F_Fecha_Respuesta_Existe/
  -- F_Fecha_Asig_Inves/F_Fecha_Respuesta_Inves (mismo patrón N+1 ya corregido en F_GetInfoGrillas).
  FUNCTION F_GetCasosControlGestion(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR;

  -- Catálogo de las 8 regiones de Policía (VM_UNIDADES.REGI_CODIGO/DESC_REGIONAL) que tienen al
  -- menos un caso IRISP1 registrado, para el filtro en cascada Región -> Sigla de unidad.
  FUNCTION F_GetRegionesIrisp1 RETURN SYS_REFCURSOR;

  -- Catálogo de siglas de unidad (IRISP_CRIMINALIDAD.SIGLA_UNIDAD) para el filtro del mapa de
  -- puntos y del tablero. Si se informa P_RegionCodigo, solo devuelve las unidades de esa región
  -- (join VM_UNIDADES.SIGLA_FISICA = IRISP_CRIMINALIDAD.SIGLA_UNIDAD) — alimenta el combo
  -- dependiente "Sigla de unidad" cuando el usuario elige una región.
  FUNCTION F_GetSiglasUnidadIrisp1(P_RegionCodigo IN NUMBER DEFAULT NULL) RETURN SYS_REFCURSOR;

  -- Resultado de cada caso IRISP1 del rango de fechas: estado general (ID_ESTADO, dominio
  -- PADRE_ID=1) y resultado de existencia (ID_ESTADO_EXISTENCIA, dominio PADRE_ID=63), con la
  -- unidad responsable. Alimenta las métricas de efectividad/uso del sistema por unidad (casos
  -- finalizados, con existencia confirmada/descartada, abiertos e inconclusos).
  FUNCTION F_GetResultadosCasosIrisp1(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR;

  -- Puntos de georreferenciación de casos IRISP1 (mapa de calor/cluster/puntos), filtrables por
  -- región, unidad (SIGLA_UNIDAD), rango de fechas y rol.
  FUNCTION F_GetMapaIrisp1(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR;

END PK_CONTROL_GESTION_IRIS;
/


CREATE OR REPLACE PACKAGE BODY PK_CONTROL_GESTION_IRIS AS

  --------------------------------------------------------------------------
  FUNCTION F_Generar_UUID RETURN NVARCHAR2 IS
    v_raw  RAW(16);
    v_uuid NVARCHAR2(36);
  BEGIN
    v_raw := SYS_GUID();
    v_uuid := REGEXP_REPLACE(LOWER(RAWTOHEX(v_raw)),
                             '(.{8})(.{4})(.{4})(.{4})(.{12})',
                             '\1-\2-\3-\4-\5');
    RETURN v_uuid;
  END F_Generar_UUID;

  --------------------------------------------------------------------------
  FUNCTION F_GetSlaConfig RETURN SYS_REFCURSOR IS
    l_rc SYS_REFCURSOR;
  BEGIN
    OPEN l_rc FOR
      SELECT
        s.SLA_CONFIG_ID       AS SlaConfigId,
        s.ID_LISTA_TAREAS     AS IdListaTareas,
        d.DESCRIPCION         AS DescTipoTarea,
        s.TIEMPO_ALERTA_HORAS AS TiempoAlertaHoras,
        s.TIEMPO_MAXIMO_HORAS AS TiempoMaximoHoras,
        s.VIGENTE             AS Vigente,
        s.JUSTIFICACION       AS Justificacion,
        s.FECHA_CREACION      AS FechaCreacion,
        s.FECHA_MODIFICA      AS FechaModifica
      FROM IRISP_SLA_CONFIG s
      LEFT JOIN IRISP_CTRL_DOMINIO d
        ON d.DOMINIO_ID = s.ID_LISTA_TAREAS
       AND d.PADRE_ID = 51
       AND d.VIGENTE = 1
      WHERE s.VIGENTE = 1
      ORDER BY d.DESCRIPCION;

    RETURN l_rc;

  EXCEPTION
    WHEN OTHERS THEN
      Pks_Administracion_IRIS.P_InsErrorsystem(
        P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.F_GetSlaConfig',
        P_Descripcion   => 'Error ' || SQLERRM,
        P_IdDocumento   => 0,
        P_Usuario       => 0
      );

      OPEN l_rc FOR
        SELECT NULL AS SlaConfigId, NULL AS IdListaTareas, NULL AS DescTipoTarea,
               NULL AS TiempoAlertaHoras, NULL AS TiempoMaximoHoras, NULL AS Vigente,
               NULL AS Justificacion, NULL AS FechaCreacion, NULL AS FechaModifica
          FROM dual WHERE 1 = 0;

      RETURN l_rc;
  END F_GetSlaConfig;

  --------------------------------------------------------------------------
  PROCEDURE P_InsUpdSlaConfig(
    P_IdListaTareas      IN NUMBER,
    P_TiempoAlertaHoras  IN NUMBER,
    P_TiempoMaximoHoras  IN NUMBER,
    P_Justificacion      IN VARCHAR2,
    P_Usuario            IN NUMBER,
    P_Maquina            IN VARCHAR2,
    P_RESULTADO          OUT NUMBER,
    SRV_Message          OUT VARCHAR2
  ) IS
    v_count         NUMBER;
    v_sla_config_id VARCHAR2(40);
  BEGIN
    IF P_IdListaTareas IS NULL THEN
      P_RESULTADO := 0;
      SRV_Message := 'Debe seleccionar el tipo de tarea';
      RETURN;
    END IF;

    IF P_TiempoAlertaHoras IS NULL OR P_TiempoMaximoHoras IS NULL THEN
      P_RESULTADO := 0;
      SRV_Message := 'Debe registrar el tiempo de alerta y el tiempo máximo';
      RETURN;
    END IF;

    IF P_TiempoAlertaHoras >= P_TiempoMaximoHoras THEN
      P_RESULTADO := 0;
      SRV_Message := 'El tiempo de alerta debe ser menor al tiempo máximo';
      RETURN;
    END IF;

    SELECT COUNT(*) INTO v_count
      FROM IRISP_SLA_CONFIG
     WHERE ID_LISTA_TAREAS = P_IdListaTareas
       AND VIGENTE = 1;

    IF v_count > 0 THEN
      UPDATE IRISP_SLA_CONFIG
         SET TIEMPO_ALERTA_HORAS     = P_TiempoAlertaHoras,
             TIEMPO_MAXIMO_HORAS     = P_TiempoMaximoHoras,
             JUSTIFICACION           = P_Justificacion,
             FECHA_MODIFICA          = SYSDATE,
             IDENTIFICACION_MODIFICA = P_Usuario,
             MAQUINA_MODIFICA        = P_Maquina
       WHERE ID_LISTA_TAREAS = P_IdListaTareas
         AND VIGENTE = 1;

      SRV_Message := 'Configuración de SLA actualizada correctamente';
    ELSE
      v_sla_config_id := F_Generar_UUID;

      INSERT INTO IRISP_SLA_CONFIG (
        SLA_CONFIG_ID, ID_LISTA_TAREAS, TIEMPO_ALERTA_HORAS, TIEMPO_MAXIMO_HORAS,
        VIGENTE, JUSTIFICACION, FECHA_CREACION, IDENTIFICACION_CREACION, MAQUINA_CREACION
      ) VALUES (
        v_sla_config_id, P_IdListaTareas, P_TiempoAlertaHoras, P_TiempoMaximoHoras,
        1, P_Justificacion, SYSDATE, P_Usuario, P_Maquina
      );

      SRV_Message := 'Configuración de SLA registrada correctamente';
    END IF;

    COMMIT;
    P_RESULTADO := 1;

  EXCEPTION
    WHEN OTHERS THEN
      ROLLBACK;
      P_RESULTADO := 0;
      SRV_Message := 'Error al guardar la configuración de SLA: ' || SQLERRM;

      BEGIN
        Pks_Administracion_IRIS.P_InsErrorsystem(
          P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.P_InsUpdSlaConfig',
          P_Descripcion   => SRV_Message,
          P_IdDocumento   => P_Usuario,
          P_Usuario       => P_Usuario
        );
      EXCEPTION
        WHEN OTHERS THEN NULL;
      END;
  END P_InsUpdSlaConfig;

  --------------------------------------------------------------------------
  PROCEDURE P_DelSlaConfig(
    P_SlaConfigId  IN VARCHAR2,
    P_Usuario      IN NUMBER,
    P_Maquina      IN VARCHAR2,
    P_RESULTADO    OUT NUMBER,
    SRV_Message    OUT VARCHAR2
  ) IS
  BEGIN
    UPDATE IRISP_SLA_CONFIG
       SET VIGENTE                 = 0,
           FECHA_MODIFICA          = SYSDATE,
           IDENTIFICACION_MODIFICA = P_Usuario,
           MAQUINA_MODIFICA        = P_Maquina
     WHERE TRIM(SLA_CONFIG_ID) = TRIM(P_SlaConfigId)
       AND VIGENTE = 1;

    IF SQL%ROWCOUNT > 0 THEN
      COMMIT;
      P_RESULTADO := 1;
      SRV_Message := 'Configuración de SLA eliminada correctamente';
    ELSE
      P_RESULTADO := 0;
      SRV_Message := 'No se encontró el registro especificado';
    END IF;

  EXCEPTION
    WHEN OTHERS THEN
      ROLLBACK;
      P_RESULTADO := 0;
      SRV_Message := 'Error al eliminar la configuración de SLA: ' || SQLERRM;

      BEGIN
        Pks_Administracion_IRIS.P_InsErrorsystem(
          P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.P_DelSlaConfig',
          P_Descripcion   => SRV_Message,
          P_IdDocumento   => P_Usuario,
          P_Usuario       => P_Usuario
        );
      EXCEPTION
        WHEN OTHERS THEN NULL;
      END;
  END P_DelSlaConfig;

  --------------------------------------------------------------------------
  FUNCTION F_GetTareasControlGestion(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR IS

    l_rc        SYS_REFCURSOR;
    V_NuevaUnde NUMBER := P_CodigoUnidad;
    V_Count     NUMBER := 0;
    V_TempUnde  NUMBER;

  BEGIN
    ------------------------------------------------------------------
    -- Escalar hasta 3 niveles de unidad si la unidad del usuario no tiene
    -- responsabilidades propias en el rango de fechas (mismo patrón que
    -- PK_VERIFICACION_IRIS.F_GetInfoGrillas / PK_CONSULTA_IRISP.F_GetInfoGrillas).
    ------------------------------------------------------------------
    FOR i IN 1..3 LOOP
      SELECT COUNT(*) INTO V_Count
        FROM IRISP_RESPON_VALIDACION c
        JOIN VM_UNIDADES f ON c.id_unidad_respon = f.consecutivo
       WHERE c.vigente = 1
         AND c.FECHA_CREACION >= P_FechaInicio
         AND c.FECHA_CREACION < (P_FechaFin + 1)
         AND NVL(f.CONSECUTIVO,0) = V_NuevaUnde;

      EXIT WHEN V_Count > 0;

      BEGIN
        V_TempUnde := TO_NUMBER(F_Consultar_UndeConsecutivo(V_NuevaUnde));
        IF V_TempUnde IS NOT NULL THEN
          V_NuevaUnde := V_TempUnde;
        ELSE
          EXIT;
        END IF;
      EXCEPTION
        WHEN OTHERS THEN EXIT;
      END;
    END LOOP;

    OPEN l_rc FOR
      WITH
      ROLES_LIST AS (
        SELECT TO_NUMBER(TRIM(REGEXP_SUBSTR(P_Roles, '[^,]+', 1, LEVEL))) AS ID_ROL
          FROM dual
        CONNECT BY LEVEL <= REGEXP_COUNT(P_Roles, '[^,]+')
      ),
      -- VM_UNIDADES trae varias filas históricas por unidad. Se deduplica a UNA sola fila por
      -- clave con ROW_NUMBER (NO con MAX(FECHA_ACTUALIZA) + join por igualdad: si dos filas
      -- vigentes empataban en esa fecha, el join devolvía 2 filas y duplicaba cada registro).
      UND_POR_CONS AS (
        SELECT consecutivo, DESCRIPCION_DEPENDENCIA, SIGLA_DEPENDE, SIGLA_FISICA, SIGLA_PAPA
          FROM (
            SELECT v.consecutivo, v.DESCRIPCION_DEPENDENCIA, v.SIGLA_DEPENDE, v.SIGLA_FISICA, v.SIGLA_PAPA,
                   ROW_NUMBER() OVER (PARTITION BY v.consecutivo ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      ),
      UND_POR_SIGLA AS (
        SELECT SIGLA_FISICA, REGI_CODIGO
          FROM (
            SELECT v.SIGLA_FISICA, v.REGI_CODIGO,
                   ROW_NUMBER() OVER (PARTITION BY v.SIGLA_FISICA ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.SIGLA_FISICA IS NOT NULL
               AND v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      )
      SELECT
        c.CRIMINALIDAD_ID         AS CriminalidadId,
        c.CODIGO                  AS Codigo,
        r.RESPON_VALIDACION_ID    AS IdResponsable,
        u.DESCRIPCION_DEPENDENCIA AS Unidad,
        u.SIGLA_DEPENDE           AS UnidadSigla,
        t.TAREA_ID                AS TareaId,
        t.ID_LISTA_TAREAS         AS IdListaTareas,
        lt.DESCRIPCION            AS DescListaTarea,
        t.ID_ESTADO_TAREA         AS IdEstadoTarea,
        et.DESCRIPCION            AS DescEstadoTarea,
        t.FECHA_CREACION          AS FechaCreacion,
        t.FECHA_MODIFICA          AS FechaModifica,
        ROUND((NVL(t.FECHA_MODIFICA, SYSDATE) - t.FECHA_CREACION) * 24, 1) AS HorasTranscurridas,
        s.TIEMPO_ALERTA_HORAS     AS TiempoAlertaHoras,
        s.TIEMPO_MAXIMO_HORAS     AS TiempoMaximoHoras,
        CASE
          WHEN s.SLA_CONFIG_ID IS NULL THEN 'SIN SLA DEFINIDO'
          WHEN (NVL(t.FECHA_MODIFICA, SYSDATE) - t.FECHA_CREACION) * 24 >= s.TIEMPO_MAXIMO_HORAS THEN 'VENCIDO'
          WHEN (NVL(t.FECHA_MODIFICA, SYSDATE) - t.FECHA_CREACION) * 24 >= s.TIEMPO_ALERTA_HORAS THEN 'EN RIESGO'
          ELSE 'A TIEMPO'
        END AS EstadoSla

      FROM IRISP_TAREA t
      JOIN IRISP_RESPON_VALIDACION r
        ON t.RESPON_VALIDACION_ID = r.RESPON_VALIDACION_ID
       AND r.VIGENTE = 1
      JOIN IRISP_CRIMINALIDAD c
        ON r.CRIMINALIDAD_ID = c.CRIMINALIDAD_ID
       AND c.VIGENTE = 1
      LEFT JOIN UND_POR_CONS u ON r.ID_UNIDAD_RESPON = u.consecutivo
      LEFT JOIN UND_POR_SIGLA ureg ON c.SIGLA_UNIDAD = ureg.SIGLA_FISICA
      LEFT JOIN IRISP_CTRL_DOMINIO lt
        ON t.ID_LISTA_TAREAS = lt.DOMINIO_ID AND lt.PADRE_ID = 51 AND lt.VIGENTE = 1
      LEFT JOIN IRISP_CTRL_DOMINIO et
        ON t.ID_ESTADO_TAREA = et.DOMINIO_ID AND et.PADRE_ID = 58 AND et.VIGENTE = 1
      LEFT JOIN IRISP_SLA_CONFIG s
        ON s.ID_LISTA_TAREAS = t.ID_LISTA_TAREAS AND s.VIGENTE = 1

      WHERE t.VIGENTE = 1
        AND t.FECHA_CREACION >= P_FechaInicio
        AND t.FECHA_CREACION <  (P_FechaFin + 1)
        AND (P_RegionCodigo IS NULL OR ureg.REGI_CODIGO = P_RegionCodigo)
        AND (P_SiglaUnidad IS NULL OR c.SIGLA_UNIDAD = P_SiglaUnidad)

        ------------------------------------------------------------
        -- FILTRO DE ROLES: idéntico al de PK_VERIFICACION_IRIS.F_GetInfoGrillas
        ------------------------------------------------------------
        AND (
                 EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL IN (1, 2))

              OR (
                   EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL = 8)
                   AND r.ID_UNIDAD_RESPON = V_NuevaUnde
                 )
            )

      ORDER BY t.FECHA_CREACION DESC;

    RETURN l_rc;

  EXCEPTION
    WHEN OTHERS THEN
      Pks_Administracion_IRIS.P_InsErrorsystem(
        P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.F_GetTareasControlGestion',
        P_Descripcion   => 'Error ' || SQLERRM,
        P_IdDocumento   => 0,
        P_Usuario       => 0
      );

      OPEN l_rc FOR
        SELECT NULL AS CriminalidadId, NULL AS Codigo, NULL AS IdResponsable, NULL AS Unidad,
               NULL AS UnidadSigla, NULL AS TareaId, NULL AS IdListaTareas, NULL AS DescListaTarea,
               NULL AS IdEstadoTarea, NULL AS DescEstadoTarea, NULL AS FechaCreacion,
               NULL AS FechaModifica, NULL AS HorasTranscurridas, NULL AS TiempoAlertaHoras,
               NULL AS TiempoMaximoHoras, NULL AS EstadoSla
          FROM dual WHERE 1 = 0;

      RETURN l_rc;
  END F_GetTareasControlGestion;

  --------------------------------------------------------------------------
  -- KPIs de tiempo de gestión a nivel de caso (creación->finalización) y por etapa
  -- (Verificación/Investigación). Reimplementa en un único query set-based la lógica de
  -- F_Fecha_Verif_Existe / F_Fecha_Respuesta_Existe / F_Fecha_Asig_Inves / F_Fecha_Respuesta_Inves
  -- (ver Oracle/Funciones.txt): por caso, toma la tarea de ese tipo creada más recientemente como
  -- inicio de etapa, y la respuesta (fecha_modifica) más reciente como fin de etapa.
  --------------------------------------------------------------------------
  FUNCTION F_GetCasosControlGestion(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR IS

    l_rc        SYS_REFCURSOR;
    V_NuevaUnde NUMBER := P_CodigoUnidad;
    V_Count     NUMBER := 0;
    V_TempUnde  NUMBER;

  BEGIN
    FOR i IN 1..3 LOOP
      SELECT COUNT(*) INTO V_Count
        FROM IRISP_RESPON_VALIDACION c
        JOIN VM_UNIDADES f ON c.id_unidad_respon = f.consecutivo
       WHERE c.vigente = 1
         AND c.FECHA_CREACION >= P_FechaInicio
         AND c.FECHA_CREACION < (P_FechaFin + 1)
         AND NVL(f.CONSECUTIVO,0) = V_NuevaUnde;

      EXIT WHEN V_Count > 0;

      BEGIN
        V_TempUnde := TO_NUMBER(F_Consultar_UndeConsecutivo(V_NuevaUnde));
        IF V_TempUnde IS NOT NULL THEN
          V_NuevaUnde := V_TempUnde;
        ELSE
          EXIT;
        END IF;
      EXCEPTION
        WHEN OTHERS THEN EXIT;
      END;
    END LOOP;

    OPEN l_rc FOR
      WITH
      ROLES_LIST AS (
        SELECT TO_NUMBER(TRIM(REGEXP_SUBSTR(P_Roles, '[^,]+', 1, LEVEL))) AS ID_ROL
          FROM dual
        CONNECT BY LEVEL <= REGEXP_COUNT(P_Roles, '[^,]+')
      ),
      RESP AS (
        SELECT rv.*
          FROM (
            SELECT r.*, ROW_NUMBER() OVER (PARTITION BY r.criminalidad_id ORDER BY r.fecha_creacion DESC) rn
              FROM IRISP_RESPON_VALIDACION r
             WHERE r.vigente = 1
          ) rv
         WHERE rv.rn = 1
      ),
      VERIF_INI AS (
        SELECT criminalidad_id, fecha_creacion FROM (
          SELECT r.criminalidad_id, it.fecha_creacion,
                 ROW_NUMBER() OVER (PARTITION BY r.criminalidad_id ORDER BY it.fecha_creacion DESC) rn
            FROM IRISP_RESPON_VALIDACION r
            JOIN IRISP_TAREA it ON it.respon_validacion_id = r.respon_validacion_id
           WHERE it.id_lista_tareas = 52 AND r.vigente = 1 AND it.vigente = 1
        ) WHERE rn = 1
      ),
      VERIF_FIN AS (
        SELECT criminalidad_id, fecha_modifica FROM (
          SELECT r.criminalidad_id, it.fecha_modifica,
                 ROW_NUMBER() OVER (PARTITION BY r.criminalidad_id ORDER BY it.fecha_modifica DESC) rn
            FROM IRISP_RESPON_VALIDACION r
            JOIN IRISP_TAREA it ON it.respon_validacion_id = r.respon_validacion_id
           WHERE it.id_lista_tareas = 52 AND it.fecha_modifica IS NOT NULL AND r.vigente = 1 AND it.vigente = 1
        ) WHERE rn = 1
      ),
      INVES_INI AS (
        SELECT criminalidad_id, fecha_creacion FROM (
          SELECT r.criminalidad_id, it.fecha_creacion,
                 ROW_NUMBER() OVER (PARTITION BY r.criminalidad_id ORDER BY it.fecha_creacion DESC) rn
            FROM IRISP_RESPON_VALIDACION r
            JOIN IRISP_TAREA it ON it.respon_validacion_id = r.respon_validacion_id
           WHERE it.id_lista_tareas = 71 AND r.vigente = 1 AND it.vigente = 1
        ) WHERE rn = 1
      ),
      INVES_FIN AS (
        SELECT criminalidad_id, fecha_modifica FROM (
          SELECT r.criminalidad_id, it.fecha_modifica,
                 ROW_NUMBER() OVER (PARTITION BY r.criminalidad_id ORDER BY it.fecha_modifica DESC) rn
            FROM IRISP_RESPON_VALIDACION r
            JOIN IRISP_TAREA it ON it.respon_validacion_id = r.respon_validacion_id
           WHERE it.id_lista_tareas = 71 AND it.fecha_modifica IS NOT NULL AND r.vigente = 1 AND it.vigente = 1
        ) WHERE rn = 1
      ),
      -- VM_UNIDADES trae varias filas históricas por unidad. Se deduplica a UNA sola fila por
      -- clave con ROW_NUMBER (NO con MAX(FECHA_ACTUALIZA) + join por igualdad: si dos filas
      -- vigentes empataban en esa fecha, el join devolvía 2 filas y duplicaba cada caso).
      UND_POR_CONS AS (
        SELECT consecutivo, DESCRIPCION_DEPENDENCIA, SIGLA_DEPENDE, SIGLA_FISICA, SIGLA_PAPA
          FROM (
            SELECT v.consecutivo, v.DESCRIPCION_DEPENDENCIA, v.SIGLA_DEPENDE, v.SIGLA_FISICA, v.SIGLA_PAPA,
                   ROW_NUMBER() OVER (PARTITION BY v.consecutivo ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      ),
      UND_POR_SIGLA AS (
        SELECT SIGLA_FISICA, REGI_CODIGO
          FROM (
            SELECT v.SIGLA_FISICA, v.REGI_CODIGO,
                   ROW_NUMBER() OVER (PARTITION BY v.SIGLA_FISICA ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.SIGLA_FISICA IS NOT NULL
               AND v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      )
      SELECT
        c.CRIMINALIDAD_ID         AS CriminalidadId,
        c.CODIGO                  AS Codigo,
        u.DESCRIPCION_DEPENDENCIA AS Unidad,
        u.SIGLA_DEPENDE           AS UnidadSigla,
        c.FECHA_CREACION          AS FechaCreacion,
        c.ID_ESTADO               AS IdEstado,
        CASE WHEN c.ID_ESTADO = 5 THEN c.FECHA_MODIFICA ELSE NULL END AS FechaFinalizacion,
        CASE WHEN c.ID_ESTADO = 5
             THEN ROUND((c.FECHA_MODIFICA - c.FECHA_CREACION) * 24, 1)
             ELSE NULL END AS HorasTotalCaso,
        vi.fecha_creacion         AS VerifInicio,
        vf.fecha_modifica         AS VerifFin,
        CASE WHEN vi.fecha_creacion IS NOT NULL AND vf.fecha_modifica IS NOT NULL
             THEN ROUND((vf.fecha_modifica - vi.fecha_creacion) * 24, 1)
             ELSE NULL END AS HorasVerificacion,
        ii.fecha_creacion         AS InvesInicio,
        invf.fecha_modifica       AS InvesFin,
        CASE WHEN ii.fecha_creacion IS NOT NULL AND invf.fecha_modifica IS NOT NULL
             THEN ROUND((invf.fecha_modifica - ii.fecha_creacion) * 24, 1)
             ELSE NULL END AS HorasInvestigacion

      FROM IRISP_CRIMINALIDAD c
      LEFT JOIN RESP r ON r.criminalidad_id = c.CRIMINALIDAD_ID
      LEFT JOIN UND_POR_CONS u ON r.ID_UNIDAD_RESPON = u.consecutivo
      LEFT JOIN UND_POR_SIGLA ureg ON c.SIGLA_UNIDAD = ureg.SIGLA_FISICA
      LEFT JOIN VERIF_INI vi ON vi.criminalidad_id = c.CRIMINALIDAD_ID
      LEFT JOIN VERIF_FIN vf ON vf.criminalidad_id = c.CRIMINALIDAD_ID
      LEFT JOIN INVES_INI ii ON ii.criminalidad_id = c.CRIMINALIDAD_ID
      LEFT JOIN INVES_FIN invf ON invf.criminalidad_id = c.CRIMINALIDAD_ID

      WHERE c.VIGENTE = 1
        AND c.FECHA_CREACION >= P_FechaInicio
        AND c.FECHA_CREACION <  (P_FechaFin + 1)
        AND (P_RegionCodigo IS NULL OR ureg.REGI_CODIGO = P_RegionCodigo)
        AND (P_SiglaUnidad IS NULL OR c.SIGLA_UNIDAD = P_SiglaUnidad)

        AND (
                 EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL IN (1, 2))

              OR (
                   EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL = 8)
                   AND r.ID_UNIDAD_RESPON = V_NuevaUnde
                 )
            )

      ORDER BY c.FECHA_CREACION DESC;

    RETURN l_rc;

  EXCEPTION
    WHEN OTHERS THEN
      Pks_Administracion_IRIS.P_InsErrorsystem(
        P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.F_GetCasosControlGestion',
        P_Descripcion   => 'Error ' || SQLERRM,
        P_IdDocumento   => 0,
        P_Usuario       => 0
      );

      OPEN l_rc FOR
        SELECT NULL AS CriminalidadId, NULL AS Codigo, NULL AS Unidad, NULL AS UnidadSigla,
               NULL AS FechaCreacion, NULL AS IdEstado, NULL AS FechaFinalizacion,
               NULL AS HorasTotalCaso, NULL AS VerifInicio, NULL AS VerifFin, NULL AS HorasVerificacion,
               NULL AS InvesInicio, NULL AS InvesFin, NULL AS HorasInvestigacion
          FROM dual WHERE 1 = 0;

      RETURN l_rc;
  END F_GetCasosControlGestion;

  --------------------------------------------------------------------------
  -- Fuente de verdad: IRISP_CRIMINALIDAD con VIGENTE=1 (los casos reales), no VM_UNIDADES —
  -- VM_UNIDADES se usa únicamente para resolver a qué región pertenece cada SIGLA_UNIDAD que ya
  -- tiene casos registrados. Antes se probó sacar el catálogo directo de VM_UNIDADES y quedó
  -- corto: una unidad con casos reales (ej. MEBAR, con cientos de registros) desaparecía del
  -- combo si la fila más reciente de VM_UNIDADES para esa sigla resultaba no vigente. La
  -- deduplicación se hace con ROW_NUMBER sobre las filas VIGENTE='SI' (una fila por sigla),
  -- evitando además el duplicado ×2 que causaba el antiguo join por MAX(FECHA_ACTUALIZA).
  FUNCTION F_GetRegionesIrisp1 RETURN SYS_REFCURSOR IS
    l_rc SYS_REFCURSOR;
  BEGIN
    OPEN l_rc FOR
      WITH
      UND_POR_SIGLA AS (
        SELECT SIGLA_FISICA, REGI_CODIGO, DESC_REGIONAL
          FROM (
            SELECT v.SIGLA_FISICA, v.REGI_CODIGO, v.DESC_REGIONAL,
                   ROW_NUMBER() OVER (PARTITION BY v.SIGLA_FISICA ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.SIGLA_FISICA IS NOT NULL
               AND v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      )
      SELECT DISTINCT u.REGI_CODIGO AS RegionCodigo, u.DESC_REGIONAL AS RegionDescripcion
        FROM IRISP_CRIMINALIDAD c
        JOIN UND_POR_SIGLA u ON c.SIGLA_UNIDAD = u.SIGLA_FISICA
       WHERE c.VIGENTE = 1
         AND c.SIGLA_UNIDAD IS NOT NULL
         AND u.REGI_CODIGO IS NOT NULL
       ORDER BY u.REGI_CODIGO;

    RETURN l_rc;

  EXCEPTION
    WHEN OTHERS THEN
      Pks_Administracion_IRIS.P_InsErrorsystem(
        P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.F_GetRegionesIrisp1',
        P_Descripcion   => 'Error ' || SQLERRM,
        P_IdDocumento   => 0,
        P_Usuario       => 0
      );

      OPEN l_rc FOR SELECT NULL AS RegionCodigo, NULL AS RegionDescripcion FROM dual WHERE 1 = 0;
      RETURN l_rc;
  END F_GetRegionesIrisp1;

  -- Ver nota de F_GetRegionesIrisp1: fuente de verdad es IRISP_CRIMINALIDAD VIGENTE=1.
  FUNCTION F_GetSiglasUnidadIrisp1(P_RegionCodigo IN NUMBER DEFAULT NULL) RETURN SYS_REFCURSOR IS
    l_rc SYS_REFCURSOR;
  BEGIN
    OPEN l_rc FOR
      WITH
      UND_POR_SIGLA AS (
        SELECT SIGLA_FISICA, REGI_CODIGO
          FROM (
            SELECT v.SIGLA_FISICA, v.REGI_CODIGO,
                   ROW_NUMBER() OVER (PARTITION BY v.SIGLA_FISICA ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.SIGLA_FISICA IS NOT NULL
               AND v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      )
      SELECT DISTINCT c.SIGLA_UNIDAD AS SiglaUnidad
        FROM IRISP_CRIMINALIDAD c
        LEFT JOIN UND_POR_SIGLA u ON c.SIGLA_UNIDAD = u.SIGLA_FISICA
       WHERE c.VIGENTE = 1
         AND c.SIGLA_UNIDAD IS NOT NULL
         AND (P_RegionCodigo IS NULL OR u.REGI_CODIGO = P_RegionCodigo)
       ORDER BY c.SIGLA_UNIDAD;

    RETURN l_rc;

  EXCEPTION
    WHEN OTHERS THEN
      Pks_Administracion_IRIS.P_InsErrorsystem(
        P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.F_GetSiglasUnidadIrisp1',
        P_Descripcion   => 'Error ' || SQLERRM,
        P_IdDocumento   => 0,
        P_Usuario       => 0
      );

      OPEN l_rc FOR SELECT NULL AS SiglaUnidad FROM dual WHERE 1 = 0;
      RETURN l_rc;
  END F_GetSiglasUnidadIrisp1;

  --------------------------------------------------------------------------
  FUNCTION F_GetResultadosCasosIrisp1(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR IS

    l_rc        SYS_REFCURSOR;
    V_NuevaUnde NUMBER := P_CodigoUnidad;
    V_Count     NUMBER := 0;
    V_TempUnde  NUMBER;

  BEGIN
    FOR i IN 1..3 LOOP
      SELECT COUNT(*) INTO V_Count
        FROM IRISP_RESPON_VALIDACION c
        JOIN VM_UNIDADES f ON c.id_unidad_respon = f.consecutivo
       WHERE c.vigente = 1
         AND c.FECHA_CREACION >= P_FechaInicio
         AND c.FECHA_CREACION < (P_FechaFin + 1)
         AND NVL(f.CONSECUTIVO,0) = V_NuevaUnde;

      EXIT WHEN V_Count > 0;

      BEGIN
        V_TempUnde := TO_NUMBER(F_Consultar_UndeConsecutivo(V_NuevaUnde));
        IF V_TempUnde IS NOT NULL THEN
          V_NuevaUnde := V_TempUnde;
        ELSE
          EXIT;
        END IF;
      EXCEPTION
        WHEN OTHERS THEN EXIT;
      END;
    END LOOP;

    OPEN l_rc FOR
      WITH
      ROLES_LIST AS (
        SELECT TO_NUMBER(TRIM(REGEXP_SUBSTR(P_Roles, '[^,]+', 1, LEVEL))) AS ID_ROL
          FROM dual
        CONNECT BY LEVEL <= REGEXP_COUNT(P_Roles, '[^,]+')
      ),
      RESP AS (
        SELECT rv.*
          FROM (
            SELECT r.*, ROW_NUMBER() OVER (PARTITION BY r.criminalidad_id ORDER BY r.fecha_creacion DESC) rn
              FROM IRISP_RESPON_VALIDACION r
             WHERE r.vigente = 1
          ) rv
         WHERE rv.rn = 1
      ),
      -- VM_UNIDADES trae varias filas históricas por unidad. Se deduplica a UNA sola fila por
      -- clave con ROW_NUMBER (NO con MAX(FECHA_ACTUALIZA) + join por igualdad: si dos filas
      -- vigentes empataban en esa fecha, el join devolvía 2 filas y DUPLICABA cada caso —causa
      -- del bug de registros ×2). ROW_NUMBER garantiza exactamente una fila por unidad.
      UND_POR_CONS AS (
        SELECT consecutivo, DESCRIPCION_DEPENDENCIA, SIGLA_DEPENDE, SIGLA_FISICA, SIGLA_PAPA
          FROM (
            SELECT v.consecutivo, v.DESCRIPCION_DEPENDENCIA, v.SIGLA_DEPENDE, v.SIGLA_FISICA, v.SIGLA_PAPA,
                   ROW_NUMBER() OVER (PARTITION BY v.consecutivo ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      ),
      UND_POR_SIGLA AS (
        SELECT SIGLA_FISICA, REGI_CODIGO
          FROM (
            SELECT v.SIGLA_FISICA, v.REGI_CODIGO,
                   ROW_NUMBER() OVER (PARTITION BY v.SIGLA_FISICA ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.SIGLA_FISICA IS NOT NULL
               AND v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      )
      SELECT
        c.CRIMINALIDAD_ID         AS CriminalidadId,
        c.CODIGO                  AS Codigo,
        -- Unidad que REGISTRA el caso (la sigla del filtro) y su dependencia/estación de policía.
        fdep.DESCRIPCION_DEPENDENCIA AS Unidad,
        c.SIGLA_UNIDAD               AS UnidadSigla,
        c.SIGLA_UNIDAD || ' - ' || NVL(fdep.DESCRIPCION_DEPENDENCIA, 'Sin dependencia') AS Dependencia,
        -- Unidad de VERIFICACIÓN (a la que se asigna la investigación) = IRISP_RESPON_VALIDACION.ID_UNIDAD_RESPON.
        -- Sigla = SIGLA_PAPA (la sigla funcional real de estas unidades investigativas, ej. DIJIN,
        -- DIPOL — la misma que usa PK_SEGUIMIENTO_IRIS.F_GetUnidadesPorSigla), para etiquetar en
        -- tablas/gráficos. La DESCRIPCION_DEPENDENCIA (ej. "SECCIONAL INVESTIGACION CRIMINAL MECAL")
        -- es el nombre completo, para mostrar en el detalle.
        uver.DESCRIPCION_DEPENDENCIA AS UnidadVerificacion,
        uver.SIGLA_PAPA              AS UnidadVerificacionSigla,
        c.FECHA_CREACION          AS FechaCreacion,
        c.ID_ESTADO               AS IdEstado,
        de.DESCRIPCION            AS DescEstado,
        c.ID_ESTADO_EXISTENCIA    AS IdEstadoExistencia,
        dx.DESCRIPCION            AS DescEstadoExistencia,
        -- Funcionario que INFORMÓ el caso (IRISP_CRIMINALIDAD.IDENTIFICACION_INFORMA): es quien
        -- está en la calle detectando el posible delito. Nombre, sigla de su unidad (SIGLA_FISICA,
        -- ej. MEBAR), dependencia y cargo se resuelven de VM_REHU_PERSONAL (identificacion es única
        -- en esa vista, así que el LEFT JOIN no multiplica filas). Si la identificación no está en
        -- la vista de personal (retirado/externo), estos campos quedan NULL.
        c.IDENTIFICACION_INFORMA  AS IdentificacionInforma,
        CASE WHEN pinf.identificacion IS NOT NULL
             THEN TRIM(pinf.grad_alfabetico || ' ' || pinf.apellidos || ' ' || pinf.nombres)
             ELSE NULL END           AS FuncionarioInforma,
        uinf.SIGLA_FISICA            AS SiglaInforma,
        pinf.descripcion_dependencia AS DependenciaInforma,
        pinf.cargo_actual            AS CargoInforma,
        -- "Tiene resultados" = el caso llegó a SPOA/SIEDCO (IRISP_RESULTADO id_tipo 69/70), misma
        -- semántica que la grilla de Verificación.
        CASE WHEN EXISTS (SELECT 1 FROM IRISP_RESULTADO rr
                           WHERE rr.criminalidad_id = c.CRIMINALIDAD_ID
                             AND rr.vigente = 1
                             AND rr.id_tipo IN (69, 70)
                             AND rr.nro_spoa_siedco IS NOT NULL) THEN 1 ELSE 0 END AS TieneResultados

      FROM IRISP_CRIMINALIDAD c
      LEFT JOIN RESP r ON r.criminalidad_id = c.CRIMINALIDAD_ID
      -- Estación/dependencia que registra el caso (c.ID_UNIDAD)
      LEFT JOIN UND_POR_CONS fdep ON c.ID_UNIDAD = fdep.consecutivo
      -- Unidad de verificación / investigación asignada (r.ID_UNIDAD_RESPON)
      LEFT JOIN UND_POR_CONS uver ON r.ID_UNIDAD_RESPON = uver.consecutivo
      LEFT JOIN UND_POR_SIGLA ureg ON c.SIGLA_UNIDAD = ureg.SIGLA_FISICA
      LEFT JOIN IRISP_CTRL_DOMINIO de ON de.DOMINIO_ID = c.ID_ESTADO AND de.PADRE_ID = 1 AND de.VIGENTE = 1
      LEFT JOIN IRISP_CTRL_DOMINIO dx ON dx.DOMINIO_ID = c.ID_ESTADO_EXISTENCIA AND dx.PADRE_ID = 63 AND dx.VIGENTE = 1
      -- Datos del funcionario que informó (identificacion es única en VM_REHU_PERSONAL: sin fan-out)
      LEFT JOIN VM_REHU_PERSONAL pinf ON pinf.identificacion = c.IDENTIFICACION_INFORMA
      -- Sigla física de la unidad donde labora el informante (ej. MEBAR), vía su unidad de trabajo
      LEFT JOIN UND_POR_CONS uinf ON pinf.unde_consecutivo_laborando = uinf.consecutivo

      WHERE c.VIGENTE = 1
        AND c.FECHA_CREACION >= P_FechaInicio
        AND c.FECHA_CREACION <  (P_FechaFin + 1)
        AND (P_RegionCodigo IS NULL OR ureg.REGI_CODIGO = P_RegionCodigo)
        AND (P_SiglaUnidad IS NULL OR c.SIGLA_UNIDAD = P_SiglaUnidad)

        AND (
                 EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL IN (1, 2))

              OR (
                   EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL = 8)
                   AND r.ID_UNIDAD_RESPON = V_NuevaUnde
                 )
            )

      ORDER BY c.FECHA_CREACION DESC;

    RETURN l_rc;

  EXCEPTION
    WHEN OTHERS THEN
      Pks_Administracion_IRIS.P_InsErrorsystem(
        P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.F_GetResultadosCasosIrisp1',
        P_Descripcion   => 'Error ' || SQLERRM,
        P_IdDocumento   => 0,
        P_Usuario       => 0
      );

      OPEN l_rc FOR
        SELECT NULL AS CriminalidadId, NULL AS Codigo, NULL AS Unidad, NULL AS UnidadSigla,
               NULL AS Dependencia, NULL AS UnidadVerificacion, NULL AS UnidadVerificacionSigla,
               NULL AS FechaCreacion, NULL AS IdEstado, NULL AS DescEstado,
               NULL AS IdEstadoExistencia, NULL AS DescEstadoExistencia,
               NULL AS IdentificacionInforma, NULL AS FuncionarioInforma,
               NULL AS SiglaInforma, NULL AS DependenciaInforma, NULL AS CargoInforma,
               NULL AS TieneResultados
          FROM dual WHERE 1 = 0;

      RETURN l_rc;
  END F_GetResultadosCasosIrisp1;

  --------------------------------------------------------------------------
  -- Mapa de puntos: reutiliza VM_GENERAL_IRISP (misma fuente que PK_REPORTES_IRIS.P_GetReporteGeneral),
  -- que ya trae latitud/longitud validadas por regex y SIGLA_UNIDAD resuelta.
  --------------------------------------------------------------------------
  FUNCTION F_GetMapaIrisp1(
    P_FechaInicio   IN DATE,
    P_FechaFin      IN DATE,
    P_RegionCodigo  IN NUMBER,
    P_SiglaUnidad   IN VARCHAR2,
    P_Roles         IN VARCHAR2,
    P_CodigoUnidad  IN NUMBER
  ) RETURN SYS_REFCURSOR IS
    l_rc    SYS_REFCURSOR;
    V_Sigla VARCHAR2(20);
  BEGIN
    BEGIN
      SELECT sig.sigla_papa INTO V_Sigla FROM VM_UNIDADES sig WHERE sig.consecutivo = P_CodigoUnidad;
    EXCEPTION WHEN NO_DATA_FOUND THEN V_Sigla := NULL;
    END;

    OPEN l_rc FOR
      WITH
      ROLES_LIST AS (
        SELECT TO_NUMBER(TRIM(REGEXP_SUBSTR(P_Roles, '[^,]+', 1, LEVEL))) AS ID_ROL
          FROM dual
        CONNECT BY LEVEL <= REGEXP_COUNT(P_Roles, '[^,]+')
      ),
      -- VM_UNIDADES trae varias filas históricas por unidad. Se deduplica a UNA sola fila por
      -- sigla con ROW_NUMBER (NO con MAX(FECHA_ACTUALIZA) + join por igualdad: si dos filas
      -- vigentes empataban en esa fecha, el join devolvía 2 filas y duplicaba cada punto).
      UND_POR_SIGLA AS (
        SELECT SIGLA_FISICA, REGI_CODIGO
          FROM (
            SELECT v.SIGLA_FISICA, v.REGI_CODIGO,
                   ROW_NUMBER() OVER (PARTITION BY v.SIGLA_FISICA ORDER BY v.FECHA_ACTUALIZA DESC) rn
              FROM VM_UNIDADES v
             WHERE v.SIGLA_FISICA IS NOT NULL
               AND v.VIGENTE = 'SI'
          )
         WHERE rn = 1
      )
      SELECT
        c.CRIMINALIDAD_ID       AS CriminalidadId,
        c.codigo_irisp          AS Codigo,
        c.SIGLA_UNIDAD          AS SiglaUnidad,
        c.MUNICIPIO             AS Municipio,
        c.BARRIO                AS Barrio,
        c.DELITO_PRINCIPAL      AS DelitoPrincipal,
        c.ESTADO                AS Estado,
        c.ESTADO_EXISTENCIA     AS EstadoExistencia,
        c.FECHA_CREACION_IRISP1 AS FechaCreacion,
        c.latitud                AS Latitud,
        c.longitud               AS Longitud
      FROM VM_GENERAL_IRISP c
      LEFT JOIN UND_POR_SIGLA ureg ON c.SIGLA_UNIDAD = ureg.SIGLA_FISICA
      WHERE REGEXP_LIKE(c.latitud, '^[0-9\.\-]+$')
        AND REGEXP_LIKE(c.longitud, '^[0-9\.\-]+$')
        AND c.FECHA_CREACION_IRISP1 >= P_FechaInicio
        AND c.FECHA_CREACION_IRISP1 <  (P_FechaFin + 1)
        AND (P_RegionCodigo IS NULL OR ureg.REGI_CODIGO = P_RegionCodigo)
        AND (P_SiglaUnidad IS NULL OR c.SIGLA_UNIDAD = P_SiglaUnidad)
        AND (
                 EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL IN (1, 2))

              OR (
                   EXISTS (SELECT 1 FROM ROLES_LIST rl WHERE rl.ID_ROL = 8)
                   AND c.SIGLA_UNIDAD = V_Sigla
                 )
            )
      ORDER BY c.FECHA_CREACION_IRISP1 DESC;

    RETURN l_rc;

  EXCEPTION
    WHEN OTHERS THEN
      Pks_Administracion_IRIS.P_InsErrorsystem(
        P_Procedimiento => 'PK_CONTROL_GESTION_IRIS.F_GetMapaIrisp1',
        P_Descripcion   => 'Error ' || SQLERRM,
        P_IdDocumento   => 0,
        P_Usuario       => 0
      );

      OPEN l_rc FOR
        SELECT NULL AS CriminalidadId, NULL AS Codigo, NULL AS SiglaUnidad, NULL AS Municipio,
               NULL AS Barrio, NULL AS DelitoPrincipal, NULL AS Estado, NULL AS EstadoExistencia,
               NULL AS FechaCreacion, NULL AS Latitud, NULL AS Longitud
          FROM dual WHERE 1 = 0;

      RETURN l_rc;
  END F_GetMapaIrisp1;

END PK_CONTROL_GESTION_IRIS;
